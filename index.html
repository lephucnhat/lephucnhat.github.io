async function callGeminiWithFallback(apiKey, preferredModel, userQuestion, isTestPing = false) { // 1. Tự động kiểm tra khóa và lấy danh sách Model khả dụng trực tiếp từ Google let candidateModels = [preferredModel || "gemini-2.5-flash", "gemini-2.5-flash", "gemini-2.5-pro"]; try { const listRes = await fetch(`https://generativelanguage.googleapis.com/v1beta/models?key=${encodeURIComponent(apiKey)}`); const listData = await listRes.json(); if (!listRes.ok) { const errMsg = listData.error?.message || `Lỗi HTTP ${listRes.status}`; return { ok: false, errorMsg: `Khóa API không hợp lệ (${listRes.status}): ${errMsg}` }; } if (Array.isArray(listData.models)) { const available = listData.models .filter(m => Array.isArray(m.supportedGenerationMethods) && m.supportedGenerationMethods.includes("generateContent")) .map(m => m.name.replace("models/", "")) .filter(name => name.startsWith("gemini-") && !name.includes("1.5")); if (available.length > 0) { const prioritized = [ preferredModel, "gemini-2.5-flash", "gemini-2.5-pro", ...available ].filter((v, i, a) => v && available.includes(v) && a.indexOf(v) === i); if (prioritized.length > 0) { candidateModels = prioritized; } } } } catch (e) { // Tiếp tục thử gọi trực tiếp nếu trình duyệt bỏ qua bước đọc danh sách } let lastError = "Không thể kết nối máy chủ Gemini."; // 2. Gọi tạo câu trả lời với hạn mức token đủ cho mô hình Thinking gemini-2.5-flash for (const modelName of candidateModels) { try { const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${encodeURIComponent(apiKey)}`; const promptText = isTestPing ? "Xin chào, hãy phản hồi ngắn gọn 1 từ: OK" : `${SYSTEM_PERSONA_PROMPT}\n\nCâu hỏi của khách truy cập: "${userQuestion}"\nHãy trả lời ấm áp, sâu sắc và chuẩn xác:`; const response = await fetch(endpoint, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ role: "user", parts: [{ text: promptText }] }], generationConfig: { temperature: 0.65, maxOutputTokens: isTestPing ? 300 : 900 } }) }); const data = await response.json(); if (response.ok && data.candidates && data.candidates.length > 0) { const parts = data.candidates[0]?.content?.parts || []; const textPart = parts.find(p => typeof p.text === "string" && !p.thought) || parts.find(p => typeof p.text === "string"); const replyText = textPart ? textPart.text.trim() : "OK"; return { ok: true, usedModel: modelName, text: replyText }; } else { const code = response.status; const apiMsg = data.error?.message || ""; if (code === 400 || code === 401 || code === 403) { return { ok: false, errorMsg: `Lỗi ${code}: ${apiMsg || "API Key không chính xác hoặc đã bị khóa."}` }; } lastError = `Lỗi ${code} (${modelName}): ${apiMsg || "Model đang bận"}`; } } catch (err) { lastError = "Lỗi kết nối mạng tới máy chủ Google Gemini."; } } return { ok: false, errorMsg: lastError }; }
